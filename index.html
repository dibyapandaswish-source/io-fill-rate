/** * Pod-wise fill rate / short supply pipeline — Munchmart (BLR & Delhi-NCR) * * This is a line-for-line TypeScript port of the Python pipeline.py that * produced the original dashboard. Keep this file and dash logic in sync — * if the date/status rules or the re-raise dedup ever change, they must * change in exactly one place (here), since both the browser (upload page) * and the server (nothing else needs it) use this same module. * * Date rules (report date D, previous day D-1): * BLR Production In-house / F&V -> D, except 7 night-dispatch CK RMs -> D-1 * all other categories -> D-1 * NCR Production In-house / F&V -> D * Dry RM -> D-1, except 4 GGN bakery items -> D * all other categories -> D-1 * Rows with Internal Order Status = 'ordered' are excluded everywhere. * * A file's report date is the LATEST expected date it contains, which is how * the export behaves when pulled on day D. City is detected from the pod * code prefix, so file names never matter. * * Re-raise dedup (BLR + Production In-house only): RMs not handed over in * their original IO get re-raised as a fresh IO and dispatched in another * slot. Both line items (same item + pod) then survive in the export, which * double-counts Ordered even though only one of them was ever shipped. Drop * the never-shipped (sent==0) duplicate(s) whenever a sibling row for the * same item+pod DID ship something. */ import Papa from 'papaparse'; export const NIGHT = [ 'Red Masala (CK)', 'Podi (CK)', 'Black Masala (CK)', 'Bhel Puri Mixture (CK)', 'Masala Peanut (CK)', 'Papdi (CK)', 'Poha Mixture (CK)', ]; export const BAKERY = [ 'Brioche Bun 4" - GGN', 'Mumbai Pav - GGN', 'Sandwich Bread - GGN', 'Tutti Frutti Bun - GGN', ]; export const TODAY_CATS = ['Production In-house', 'F&V']; export const TERMINAL = ['closed', 'received', 'force_closed']; export const CAT_ORDER = [ 'Production In-house', 'Dry RM', 'Packaging', 'Sauces And Syrups', 'Beverage', 'F&V', 'Chilled & Frozen', 'Desserts', ]; export const NCR_PREFIX = ['GGN', 'NOI', 'DEL', 'GZB']; export type City = 'BLR' | 'NCR'; export interface Row { item: string; cat: string; uom: string; pod: string; status: string; exp: string; // dd-mm-yyyy io: string; ord: number; sent: number; recd: number; city: City; } interface RuleRow extends Row { short: number; terminal: boolean; } export interface Agg { date: string; pods: string[]; pod_cat: Array<[City, string, number, number, number, number, number, number, number]>; items: Array<{ c: City; g: string; i: string; u: string; o: number; s: number; r: number; sh: number; f: number; np: number; z: number; fp: number; sp: number; p: string; pr?: Array<[number, number, number, number]>; }>; cat: Array<[City, string, number, number, number, number, number, number, number, number, number, number]>; schema: { pod_cat: string[]; cat: string[] }; } export interface Payload { dates: string[]; detail_dates: string[]; trend: any[]; detail: Record; cat_order: string[]; schema: { pod_cat: string[]; cat: string[] }; } // ---------------------------------------------------------------- helpers function r2(x: number): number { return Math.round(x * 100) / 100; } function toNum(v: unknown): number { const n = typeof v === 'string' ? parseFloat(v) : Number(v); return Number.isFinite(n) ? n : 0; } /** 'dd-mm-yyyy' -> sortable 'yyyy-mm-dd' */ export function dkey(s: string): string { const [dd, mm, yy] = s.split('-'); return `${yy}-${mm}-${dd}`; } export function prevDay(s: string): string { const [dd, mm, yy] = s.split('-').map(Number); const dt = new Date(Date.UTC(yy, mm - 1, dd)); dt.setUTCDate(dt.getUTCDate() - 1); const p = (n: number) => String(n).padStart(2, '0'); return `${p(dt.getUTCDate())}-${p(dt.getUTCMonth() + 1)}-${dt.getUTCFullYear()}`; } function groupBy(arr: T[], keyFn: (t: T) => string): Map { const m = new Map(); for (const x of arr) { const k = keyFn(x); const g = m.get(k); if (g) g.push(x); else m.set(k, [x]); } return m; } const K = (...parts: string[]) => parts.join('::'); // ---------------------------------------------------------------- loading /** * Parse one raw Internal Order CSV export. Throws if a required column is * missing (mirrors the Python pipeline's hard failure, so a malformed file * is caught immediately instead of silently producing wrong numbers). */ export function loadExport(csvText: string, filename: string): Row[] { const clean = csvText.replace(/^﻿/, ''); const parsed = Papa.parse>(clean, { header: true, skipEmptyLines: true }); const fields = parsed.meta.fields || []; const need = ['Ordered Quantity', 'Sent Quantity', 'Received Quantity']; for (const c of need) { if (!fields.includes(c)) { throw new Error(`${filename}: missing column "${c}"`); } } const out: Row[] = []; for (const r of parsed.data) { const pod = r['Internal Order From Location']; const exp = r['Internal Order Expected Date']; if (!pod || !exp) continue; const prefix = pod.split('-')[0]; const city: City = NCR_PREFIX.includes(prefix) ? 'NCR' : 'BLR'; out.push({ item: r['Product Full Name'] || '', cat: r['Category'] || '', uom: r['Unit Of Measure'] || '', pod, status: r['Internal Order Status Name'] || '', exp, io: r['Internal Order Number'] || '', ord: toNum(r['Ordered Quantity']), sent: toNum(r['Sent Quantity']), recd: toNum(r['Received Quantity']), city, }); } return out; } /** The report date a file represents: the latest expected date it contains. */ export function discoverDate(rows: Row[]): string { let best: string | null = null; for (const r of rows) { if (!best || dkey(r.exp) > dkey(best)) best = r.exp; } if (!best) throw new Error('no usable rows (missing pod/expected-date on every line)'); return best; } // ---------------------------------------------------------------- rules function dedupeReraised(rows: RuleRow[]): RuleRow[] { const groups = groupBy( rows.filter(r => r.city === 'BLR' && r.cat === 'Production In-house'), r => K(r.item, r.pod), ); const drop = new Set(); groups.forEach(g => { if (g.length > 1) { const hasSent = g.some(r => r.sent > 0); const hasZero = g.some(r => r.sent === 0); if (hasSent && hasZero) g.filter(r => r.sent === 0).forEach(r => drop.add(r)); } }); return rows.filter(r => !drop.has(r)); } function applyRules(rows: Row[], D: string): RuleRow[] { const DM1 = prevDay(D); const want = (r: Row): string => { if (TODAY_CATS.includes(r.cat)) { return r.city === 'BLR' && NIGHT.includes(r.item) ? DM1 : D; } if (r.city === 'NCR' && BAKERY.includes(r.item)) return D; return DM1; }; const filtered = rows .filter(r => r.exp === want(r) && r.status !== 'ordered') .map(r => ({ ...r, short: Math.max(r.ord - r.sent, 0), terminal: TERMINAL.includes(r.status) })); return dedupeReraised(filtered); } function classify(sub: Array<{ sent: number; fill: number | null }>) { const n = sub.length; const zero = sub.filter(s => s.sent === 0).length; const full = sub.filter(s => s.fill != null && s.fill >= 0.995).length; const fills = sub.map(s => s.fill).filter((f): f is number => f != null); let spread = 0; if (n > 1 && fills.length > 1) { const mean = fills.reduce((a, b) => a + b, 0) / fills.length; const variance = fills.reduce((a, b) => a + (b - mean) ** 2, 0) / (fills.length - 1); spread = Math.sqrt(Math.max(variance, 0)); } const sentSum = sub.reduce((a, s) => a + s.sent, 0); let pat: string; if (sentSum === 0) pat = 'total stock-out'; else if (full === n) pat = 'fully supplied'; else if (n < 2) pat = 'single pod'; else if (zero === 0 && spread < 0.12) pat = 'pro-rata cut'; else pat = 'uneven'; return { zero, full, spread, pat }; } // ---------------------------------------------------------------- aggregation /** Build one report date's aggregate. `framesRows` = one Row[] per input file. */ export function buildAgg(D: string, framesRows: Row[][]): Agg { const raw = framesRows.flat(); const d = applyRules(raw, D); if (d.length === 0) throw new Error(`no rows survived the date/status rules for ${D}`); type Line = { city: City; cat: string; item: string; uom: string; pod: string; ord: number; sent: number; recd: number; short: number; fill: number | null }; const lineMap = groupBy(d, r => K(r.city, r.cat, r.item, r.uom, r.pod)); const line: Line[] = []; lineMap.forEach((rows, key) => { const [city, cat, item, uom, pod] = key.split('::') as [City, string, string, string, string]; const ord = rows.reduce((a, r) => a + r.ord, 0); const sent = rows.reduce((a, r) => a + r.sent, 0); const recd = rows.reduce((a, r) => a + r.recd, 0); const short = rows.reduce((a, r) => a + r.short, 0); line.push({ city, cat, item, uom, pod, ord, sent, recd, short, fill: ord > 0 ? sent / ord : null }); }); const pods = Array.from(new Set(line.map(l => l.pod))).sort(); const pidx = new Map(pods.map((p, i) => [p, i])); const pcMap = groupBy(line, l => K(l.city, l.cat, l.pod)); const pod_cat: Agg['pod_cat'] = []; pcMap.forEach((rows, key) => { const [city, cat] = key.split('::') as [City, string]; const ord = rows.reduce((a, r) => a + r.ord, 0); const sent = rows.reduce((a, r) => a + r.sent, 0); const recd = rows.reduce((a, r) => a + r.recd, 0); const short = rows.reduce((a, r) => a + r.short, 0); const zi = rows.filter(r => r.ord > 0 && r.sent === 0).length; const s7 = rows.filter(r => r.ord > 0 && r.fill != null && r.fill < 0.70).length; const pod = rows[0].pod; pod_cat.push([city, cat, pidx.get(pod)!, r2(ord), r2(sent), r2(recd), r2(short), zi, s7]); }); const itemMap = groupBy(line, l => K(l.city, l.cat, l.item, l.uom)); const items: Agg['items'] = []; itemMap.forEach((rows, key) => { const [city, cat, item, uom] = key.split('::') as [City, string, string, string]; const o = rows.reduce((a, r) => a + r.ord, 0); if (o <= 0) return; const s = rows.reduce((a, r) => a + r.sent, 0); const recdSum = rows.reduce((a, r) => a + r.recd, 0); const shSum = rows.reduce((a, r) => a + r.short, 0); const { zero, full, spread, pat } = classify(rows.map(r => ({ sent: r.sent, fill: r.fill }))); const rec: Agg['items'][number] = { c: city, g: cat, i: item, u: uom, o: r2(o), s: r2(s), r: r2(recdSum), sh: r2(shSum), f: Math.round((s / o) * 10000) / 10000, np: rows.length, z: zero, fp: full, sp: Math.round(spread * 1000) / 1000, p: pat, }; rec.pr = rows .map((r): [number, number, number, number] => [pidx.get(r.pod)!, r2(r.ord), r2(r.sent), r2(r.recd)]) .sort((a, b) => { const fa = a[1] ? a[2] / a[1] : 9; const fb = b[1] ? b[2] / b[1] : 9; if (fa !== fb) return fa - fb; return b[1] - a[1]; }); items.push(rec); }); items.sort((a, b) => b.sh - a.sh); const csMap = groupBy(line, l => K(l.city, l.cat)); const cat: Agg['cat'] = []; csMap.forEach((rows, key) => { const [city, catName] = key.split('::') as [City, string]; const ord = rows.reduce((a, r) => a + r.ord, 0); const sent = rows.reduce((a, r) => a + r.sent, 0); const recd = rows.reduce((a, r) => a + r.recd, 0); const short = rows.reduce((a, r) => a + r.short, 0); const itemsN = new Set(rows.map(r => r.item)).size; const podsN = new Set(rows.map(r => r.pod)).size; const dScope = d.filter(r => r.city === city && r.cat === catName); const term = dScope.filter(r => r.terminal); const open = dScope.filter(r => !r.terminal); const t_ord = term.reduce((a, r) => a + r.ord, 0); const t_sent = term.reduce((a, r) => a + r.sent, 0); const t_recd = term.reduce((a, r) => a + r.recd, 0); const open_ord = open.reduce((a, r) => a + r.ord, 0); cat.push([city, catName, r2(ord), r2(sent), r2(recd), r2(short), itemsN, podsN, r2(t_ord), r2(t_sent), r2(t_recd), r2(open_ord)]); }); return { date: D, pods, pod_cat, items, cat, schema: { pod_cat: ['city', 'cat', 'pod', 'ord', 'sent', 'recd', 'short', 'zero_items', 'sub70'], cat: ['city', 'cat', 'ord', 'sent', 'recd', 'short', 'items', 'pods', 't_ord', 't_sent', 't_recd', 'open_ord'], }, }; } /** Equivalent of the Python main()'s trend/detail assembly. */ export function buildPayload(have: Record, detailDays: number): Payload { const dates = Object.keys(have).sort((a, b) => (dkey(a) < dkey(b) ? -1 : dkey(a) > dkey(b) ? 1 : 0)); if (dates.length === 0) throw new Error('no data'); const trend: any[] = []; dates.forEach(D => { have[D].cat.forEach(row => trend.push([D, ...row])); }); const detail_dates = dates.slice(-detailDays); const detail: Record = {}; detail_dates.forEach(D => { detail[D] = have[D]; }); return { dates, detail_dates, trend, detail, cat_order: CAT_ORDER, schema: have[dates[dates.length - 1]].schema, }; }
